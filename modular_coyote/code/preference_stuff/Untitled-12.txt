	// ckey = list(slots)
	var/list/sources = list()

	var/list/elligable_targets = list()
	for(var/ckey in GLOB.directory)
		var/client/C = GLOB.directory[ckey]
		if(!C)
			continue
		var/mob/living/current_mob = C.mob
		if(!isliving(current_mob))
			continue
		if(!safe_read_pref(current_mob, /datum/preference/toggle/rumors/hear_rumors))
			continue
		elligable_targets |= ckey

	var/list/truly_elligable_targets = elligable_targets - rm_targets_used
	if(!LAZYLEN(truly_elligable_targets))
		rm_targets_used = list()
	else
		elligable_targets = truly_elligable_targets

	var/do_offline = prob(rm_pick_offline_chance)
	var/do_nethack = !do_offline && prob(rm_chance_to_just_be_some_silly_net_hack_rumor)
	var/screwed_it = FALSE
	screw_it_do_nethack:

	if(do_nethack)
		goto do_some_nethack_rumors
	else
		var/tried_again = FALSE
		try_again:
		for(var/ckey in rumors)
			if(rm_sources_used[ckey] > world.time)
				continue
			var/online_slot = get_online_character_slot(ckey)
			var/datum/rumor_holder/rholder = GetRumorHolder(ckey)
			for(var/i in 1 to LAZYLEN(rholder.rumor_slots))
				if(do_offline)
					if(online_slot == i)
						continue
				else
					if(online_slot != i)
						continue
				if(!LAZYLEN(rholder.rumor_slots[i]))
					continue
				var/list/rumors = rholder.get_all_rumors(i, FALSE)
				if(!LAZYLEN(rumors))
					continue
				var/char_slot = i
				if(!islist(sources[ckey]))
					sources[ckey] = list()
				sources[ckey] |= char_slot
				rm_sources_used[ckey] = world.time + rm_source_stale_time
		if(!LAZYLEN(sources))
			if(!tried_again)
				rm_sources_used = list()
				tried_again = TRUE
				goto try_again
			else
				return

	var/list/lucky_targets = list()
	if(LAZYLEN(elligable_targets) <= rm_target_batch_size)
		lucky_targets = elligable_targets
	else
		for(var/i in 1 to rm_target_batch_size)
			var/ckey = pick(elligable_targets)
			lucky_targets |= ckey
			rm_targets_used |= ckey
			elligable_targets -= ckey

	// ckey = list(slots)
	do_some_nethack_rumors:
	var/list/lucky_sources = list()
	if(do_nethack)
		goto do_some_nethack_rumors_only
	var/moretries = 20
	while(moretries > 0)
		moretries -= 1
		var/ckey = pick(sources)
		var/char_slot = pick(sources[ckey])
		// do they have any rumors in this slot?
		var/datum/rumor_holder/rholder = GetRumorHolder(ckey)
		if(!LAZYLEN(rholder.rumor_slots[char_slot]))
			sources[ckey] -= char_slot
			if(!LAZYLEN(sources[ckey]))
				sources -= ckey
			continue
		if(!islist(lucky_sources[ckey]))
			lucky_sources[ckey] = list()
		lucky_sources[ckey] |= char_slot

	// if we have no sources, just pick a random nethack rumor and send it to a random target
	if(!LAZYLEN(lucky_sources))
		if(screwed_it || do_nethack)
			CRASH("RumorMill: No sources to spread rumors from, and no nethack rumors to send either! This is bad!")
		screwed_it = TRUE
		do_nethack = TRUE
		goto screw_it_do_nethack

	do_some_nethack_rumors_only:
	var/list/used_condoms = list()
	var/list/used_bottoms = list()
	var/succeeded = FALSE

	// send em!
	for(var/winnerckey in lucky_targets)
		// should be online
		if(do_nethack)
			var/datum/rumor/nethack_rumor = get_random_nethack_rumor(FALSE, TRUE)
			send_rumor_to_target(winnerckey, nethack_rumor)
			succeeded = TRUE
			continue
		var/horny = HORNYMODE_NONE
		if(safe_read_pref(winnerckey, /datum/preference/toggle/rumors/hear_horny_rumors))
			horny = HORNYMODE_EITHEROR
			if(prob(50))
				horny = HORNYMODE_ONLYHORNY
		var/datum/rumor/really_cool_rumor
		for(var/ckey in lucky_sources) // prioritize the top
			if(!debug && ckey == winnerckey)
				continue // yeah we already know about them
			var/char_slot = pick(lucky_sources[ckey])
			var/datum/rumor_holder/source_rholder = GetRumorHolder(ckey)
			try_again:
			really_cool_rumor = source_rholder.get_a_rumor(char_slot, horny, FALSE, TRUE)
			if(!really_cool_rumor)
				if(horny == HORNYMODE_ONLYHORNY)
					horny = HORNYMODE_EITHEROR
					goto try_again
				else
					continue
			if(really_cool_rumor)
				send_rumor_to_target(winnerckey, really_cool_rumor)
				if(!islist(used_condoms[winnerckey]))
					used_condoms[winnerckey] = list()
				used_condoms[winnerckey][really_cool_rumor.rumor_uid] = world.time + rm_source_stale_time
				used_bottoms[ckey] = world.time + rm_source_stale_time
				succeeded = TRUE
				break
	if(succeeded && !do_nethack)
		rm_rumors_used |= used_condoms
		rm_sources_used |= used_bottoms
		if(debug_stats)
			var/list/titters = list()
			titters["now"] = time2text(world.realtime)
			titters["rumors_targets_queued"] = LAZYLEN(rm_targets_queued)
			titters["rumors_targets_used"] = LAZYLEN(rm_targets_used)
			titters["rumors_sources_used"] = LAZYLEN(rm_sources_used)
			output_stats(titters, "SPREAD_STATS")
		if(debug)
			output_debug("RumorMill: Spread rumors to [LAZYLEN(lucky_targets)] lucky targets from [LAZYLEN(lucky_sources)] lucky sources!", TRUE)
